MAGNUM_ECOM:
LOAD
  ECOM_DK&'_'&ECOM_KASPI_ID as ID,
  
  ECOM_DK,
  ECOM_KASPI_ID,
  Date(ECOM_DATE) as ECOM_DATE,
  ECOM_NUM_DOG,
  ECOM_ACCOUNT_DT,
  ECOM_ANALYTICS_DT1,
  ECOM_ANALYTICS_DT2,
  ECOM_ACCOUNT_KT,
  ECOM_ANALYTICS_KT1,
  ECOM_ANALYTICS_KT2,
  ECOM_AMOUNT,
  ECOM_CURRENCY,
  ECOM_BIN;
SQL
WITH ECOM AS (
	SELECT *
	  FROM (
		  SELECT *, ROW_NUMBER() OVER (PARTITION BY KASPI_ID ORDER BY S$CHANGE_DATE DESC) AS ROW_NUM
		  FROM S0234.KASPI_PAYMENT_CONTROL) T
	WHERE ROW_NUM = 1)
	  
SELECT 'MagnumEcom' as ECOM_DK,
	   IF(NULL_OR_EMPTY(ECOM.KASPI_ID) = 1, '-', ECOM.KASPI_ID) AS ECOM_KASPI_ID,
	   ECOM.PERIOD as ECOM_DATE,
	   ECOM.NOMER_DOCUMENTA as ECOM_NUM_DOG,
	   ECOM.SCHET_DEBET as ECOM_ACCOUNT_DT,
	   ECOM.ANALYTIKA_DT1 as ECOM_ANALYTICS_DT1,
	   ECOM.ANALYTIKA_DT2 as ECOM_ANALYTICS_DT2,
	   -- DEBET_AMOUNT,,
	   -- DEBET_CURRENCY
	   ECOM.SCHET_CREDIT as ECOM_ACCOUNT_KT,
	   ECOM.ANALYTIKA_KT1 as ECOM_ANALYTICS_KT1,
	   ECOM.ANALYTIKA_KT2 as ECOM_ANALYTICS_KT2,
	   ECOM.CREDIT as ECOM_AMOUNT,
	   ECOM.VALYUTA as ECOM_CURRENCY,
	   ECOM.BIN as ECOM_BIN
  FROM ECOM
 WHERE DATE(ECOM.PERIOD) >= DATE '2025-01-01';
        
        
OUTER JOIN (MAGNUM_ECOM)
LOAD
  IBSO_DK&'_'&IBSO_KASPI_ID as ID,
  
  IBSO_DK,
  IBSO_KASPI_ID,
  Date(IBSO_DATE_REQ) as IBSO_DATE_REQ,
  IBSO_KHE_DK,
  IBSO_STATE,
  IBSO_AMOUNT,
  IBSO_CURRENCY,
  IBSO_INIT_NAME,
  IBSO_NAZN,
  IBSO_SUPPLIER_BIN,
  IBSO_SUPPLIER_NAME;
  
SQL	  
SELECT 'MagnumEcom' as IBSO_DK,
	   IF(NULL_OR_EMPTY(IBS.NUM) = 1, '-', IBS.NUM) AS IBSO_KASPI_ID,
	   DATE(IBS.DATE_REQ) AS IBSO_DATE_REQ,
	   -- НАИМЕНОВАНИЕ ПОСТАВЩИКА,
	   7777 as IBSO_KHE_DK,
	   IBS.STATE_ID as IBSO_STATE,
	   IBS.SUMM_PAY as IBSO_AMOUNT,
	   M.C_CUR_SHORT as IBSO_CURRENCY,
	   IBS.INIT_NAME as IBSO_INIT_NAME,
	   IBS.NAZN_PLAT as IBSO_NAZN,
	   IBS.SUPPLIER_IN as IBSO_SUPPLIER_BIN,
       IBS.SUPPLIER_NAME as IBSO_SUPPLIER_NAME
  FROM BI_VIEW.S06_Z_KAS_REQUEST_DK IBS
  JOIN S06.Z_FT_MONEY M 
    ON IBS.VALUTA = CAST(M.ID AS VARCHAR(30))
 WHERE IBS.TYPE_DK = '166405524307'
   AND DATE(DATE_REQ) >= DATE '2025-01-01';
   
NoConcatenate

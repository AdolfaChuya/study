LET varMinDate = Num(MakeDate(2010,01,1));
LET varMaxDate = Floor(num(Today()-1));
LET vYesterday = Floor(Today() - 1);

TemporaryCalendar:
LOAD 
	Date($(varMinDate) + IterNo() - 1) as TempDate
	AutoGenerate 1
	While $(varMinDate) + IterNo() - 1 <= $(varMaxDate);

MasterCalendar:
LOAD
	TempDate as "Дата",
    Date(MonthStart(TempDate), 'MMM.YY') as "Месяц+Год",
	Year(TempDate) as "Год",
    Month(TempDate) as "Месяц",
	if(Floor(TempDate) = Floor(MonthEnd(TempDate)), 1, 
		if(Floor(TempDate) = $(varMaxDate), 1)) as "Конец месяца",
	Week(TempDate, 0, 1, 1) as "Неделя",
    if(Floor(TempDate) = Floor(WeekEnd(TempDate)), 1, 
		if(Floor(TempDate) = $(varMaxDate), 1)) as "Конец недели"
RESIDENT TemporaryCalendar;

SELECT A.* FROM
(
SELECT A.*
,ROW_NUMBER() OVER (PARTITION BY SmetKey ORDER BY C_SUMMA2 DESC) AS RMM

FROM
(SELECT A.*, 'SMETSUM' as FROMW

FROM(
SELECT A.*
,ROW_NUMBER() OVER (PARTITION BY SPLR2,C_ESTIMATE_SUM ORDER BY SPLR2, C_ESTIMATE_SUM, C_DATE_REQ) AS RM
,UPPER(SPLR2)||C_ESTIMATE_SUM AS SmetKey
,COUNT(distinct C_OBJECT) OVER (PARTITION BY SPLR2||C_ESTIMATE_SUM) AS QOBG

FROM(
SELECT A.*,

CASE
WHEN SPLR LIKE '% % %'
THEN DWH_SAN.STR_P.EXTRACT(SPLR,CHR(32),CHR(32))
WHEN SPLR LIKE '% %'
THEN SUBSTR(SPLR,INSTR(SPLR,' '))
ELSE SPLR 
END as SPLR2

FROM
(SELECT  A1.C_DATE_REQ C_DATE_REQ --ДАТА ЗАЯВКИ
      , C2.C_CODE C_KHE_REQ --КХЕ СЧЕТА
      , C3.C_CODE C_KHE_BUD --КХЕ БЮДЖЕТА
      , C4.C_CODE_EXPENSE C_COST_CODE --КОД ЗАТРАТ
      , C4.C_NAME_EXPENSE C_COST_NAME --НАИМЕНОВАНИЕ ЗАТРАТ
      , A1.C_NUM C_REQ_NUM --НОМЕР ЗАЯВКИ
      
      , CASE WHEN NOT REGEXP_LIKE(C4.C_CODE_EXPENSE,'^2850') AND NVL(A1.C_NDS,'0') = '1'
             THEN (B1.C_SUMMA/100)*(100+12)
               ELSE B1.C_SUMMA END C_SUMMA --СУММА ПЛАТЕЖА (С НДС)
      ///
      ,Round(C_SUMMA) C_SUMMA2
      ,A1.C_NDS
      ,TRIM(UPPER(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(A1.C_SUPPLIER,'"',''),CHR(39),''),'',''),'«',''),'»',''),CHR(10),'')))  AS SPLR
      ///
      , A1.C_SUPPLIER C_SUPPLIER--НАИМЕНОВАНИЕ ПОСТАВЩИКА/МОЛ
      , A1.C_NAZN C_NAZN --НАЗНАЧЕНИЕ ПЛАТЕЖА
      , CASE WHEN A1.C_IN_BUDGET = '1' THEN 'Да' ELSE 'Нет' END  C_IN_BUDGET --В БЮДЖЕТЕ
      , DWH_SAN.STR_P.EXTRACT(A1.C_DOP_REC_1C, '<Branch>','</Branch>') AS C_OBJECT
      , Round(TO_NUMBER(REPLACE(REPLACE(DWH_SAN.STR_P.EXTRACT(A1.C_DOP_REC_1C, '<Amount>','</Amount>'),' ',''),',','.'))) AS C_ESTIMATE_SUM
     , DWH_SAN.STR_P.EXTRACT(A1.C_DOP_REC_1C, '<Description>', '</Description>') AS C_DISCRIPTION
     
      , A3.C_NAME C_INIT_NAME -- ИНИЦИАТОР ЗАЯВКИ
  FROM S06.Z#SPR_CODE_EXPENSE C4
     , S06.Z#HOZED C3
     , S06.Z#HOZED C2
     , S06.Z#TMC_RASH_STATS C1
     , S06.Z#TMC_DOG_LIST B1
     , S06.Z#USER A3
     , S06.Z#PRODUCT A2
     , S06.Z#KAS_REQUEST_PAY A1
  WHERE A1.ID=A2.ID AND A2.C_CREATE_USER=A3.ID(+) AND C1.C_KXE=C2.ID(+) AND C1.C_KXE_2=C3.ID(+) AND C1.C_COSTS_CODE=C4.ID(+)
    AND (A1.STATE_ID = 'PAID'
  /*  AND A1.C_DATE_REQ >= IN_DATEBEG
    AND A1.C_DATE_REQ <= IN_DATEEND+1 */
    AND A1.C_REQ_ID_1C IS NOT NULL
    AND A1.C_DOP_REC_1C IS NOT NULL
    AND B1.C_REQUEST_PAY = A1.ID
    AND C1.C_REQUEST_PAY_ARR = B1.COLLECTION_ID
    
  // AND A1.C_SUPPLIER NOT LIKE '%П%Кадыро%'
)
  ORDER BY A1.C_DATE_REQ,C3.C_CODE) A
  
  WHERE Round(C_ESTIMATE_SUM)<>Round(C_SUMMA2)
  AND C_COST_CODE IN ('2301','2303','2304','2850')
  
) A) A
WHERE RM = 1

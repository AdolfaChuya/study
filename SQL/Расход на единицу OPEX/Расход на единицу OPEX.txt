// ============================================================
// РАСХОД НА ЕДИНИЦУ (RPU) — поток Kaspi Сервис
// Заказчик: Иванова Айяна, Kaspi Сервис
// Исполнитель: Толыбаев Нурсултан
// ------------------------------------------------------------
// ВЕРСИЯ: 1240 = проводки 1240 ∪ 1245 + амортизация КХЕ 1245
//   (ПЦН = 1245/2131/ТОО "Жедел Кузет"; Тех охрана = остальное 1245),
//   витрина RPU_6M — свод a la пример заказчика «Расход на единицу 6M».
// + По требованию заказчика метрика «Расход на охрану 1 АТМ» (ПЦН)
//   вынесена ИЗ блока 1240 В ОТДЕЛЬНЫЙ КХЕ 1245
//   (переназначение только в витрине RPU; ядро FACT_* под 1240 не тронуто).
//   Контроль 1245: П1 = 100 690 | П2 = 86 249 | % = −14%.
// КОНТРОЛЬ янв-2025 (эталон):
//   1240 объекты=128 | Аморт=18 001 218 | Физ охрана=86 790 438
//   Тех охрана=10 654 829 | Стоянки=9 353 650 | Командировки=962 081
//   ПЦН=41 450 529 | Всего OPEX=149 211 527
//   RPU объекты=982 517 | RPU АТМ (1245)=12 568
// ============================================================
SET ThousandSep=' ';
SET DecimalSep=',';
SET MoneyThousandSep=' ';
SET MoneyDecimalSep=',';
SET MoneyFormat='$#,##0.00;-$#,##0.00';
SET TimeFormat='hh:mm:ss';
SET DateFormat='DD.MM.YYYY';
SET TimestampFormat='DD.MM.YYYY hh:mm:ss[.fff]';
SET FirstWeekDay=0;
SET BrokenWeeks=1;
SET ReferenceDay=0;
SET FirstMonthOfYear=1;
SET CollationLocale='ru-RU';
SET MonthNames='Янв;Фев;Мар;Апр;Май;Июн;Июл;Авг;Сен;Окт;Ноя;Дек';
SET LongMonthNames='Январь;Февраль;Март;Апрель;Май;Июнь;Июль;Август;Сентябрь;Октябрь;Ноябрь;Декабрь';
SET DayNames='Пн.;Вт.;Ср.;Чт.;Пт.;Сб.;Вс.';
SET LongDayNames='Понедельник;Вторник;Среда;Четверг;Пятница;Суббота;Воскресенье';
SET NumericalAbbreviation='3:k;6:M;9:G;12:T;15:P;18:E;21:Z;24:Y;-3:m;-6:μ;-9:n;-12:p;-15:f;-18:a;-21:z;-24:y';

// 1270 — Кол-во банкнот (годовые QVD по маске)

// ============================================================
// 1250 — Кол-во поездок
// ВНИМАНИЕ: QVD статичен (данные по июнь-2026; июля-августа нет).
// Решение об источнике (перегенерация QVD или StarRocks BI.F$INCASH_DEP)
// ещё не принято — блок оставлен как есть.
// ============================================================
RAW_1250:
LOAD
    Год, Месяц, День,
    otd_fil_visits, filial_visits, nbrk_visits,
    Text(visits) as visits,
    out_atm_visits, out_trm_visits, in_trm_visits, in_atm_visits,
    CLNT_SUM_CNT
FROM [lib://QVD Tables/Incass/QVD_Incash_20250101.qvd] (qvd);

KOL_INCASH_1250:
LOAD
    Год, Месяц,
    1250 as КХЕ,
    Date(MakeDate(Год, Месяц), 'MMM-YYYY') as МесяцГод,
    MakeDate(Год, Месяц) as ДатаСортировки,
    Sum(otd_fil_visits) + Sum(filial_visits) + Sum(nbrk_visits)
      + Count(DISTINCT visits)
      + Sum(out_atm_visits) + Sum(out_trm_visits)
      + Sum(in_trm_visits) + Sum(in_atm_visits)
      + Sum(CLNT_SUM_CNT)                     as [Кол-во поездок]
RESIDENT RAW_1250
GROUP BY Год, Месяц;
DROP TABLE RAW_1250;

// ============================================================
// 902 — Кол-во минут тех.поддержки (HR календарь Oracle)
// ============================================================
LIB CONNECT TO 'DWH OLE DB (hq_urymbayev_28583)';

HR_CAL_902:
LOAD
    IS_DAY_OFF_FLG2,
    CALENDAR_DATE2,
    Year(CALENDAR_DATE2)  as Год,
    Month(CALENDAR_DATE2) as Месяц
;
SELECT
    IS_DAY_OFF_FLG  as IS_DAY_OFF_FLG2,
    CALENDAR_DATE   as CALENDAR_DATE2
FROM BI.HR_CALENDAR_V
WHERE CALENDAR_DATE >= DATE'2025-01-01'
  AND CALENDAR_DATE < TRUNC(SYSDATE);

KOL_902MIN:
LOAD
    Год, Месяц,
    902 as КХЕ,
    Date(MakeDate(Год, Месяц),'MMM-YYYY') as МесяцГод,
    MakeDate(Год, Месяц) as ДатаСортировки,
      Count(DISTINCT If(IS_DAY_OFF_FLG2='Y', CALENDAR_DATE2)) * 8  * 60 * 21
    + Count(DISTINCT If(IS_DAY_OFF_FLG2='N', CALENDAR_DATE2)) * 11 * 60 * 21
        as [Кол-во минут]
RESIDENT HR_CAL_902
GROUP BY Год, Месяц;
DROP TABLE HR_CAL_902;

RAW_MANUAL:
LOAD
    Year(Месяц)   as Год,
    Month(Месяц)  as Месяц,

(ooxml, embedded labels)

  AND Trim([Кол-во] & '') <> '-';

KOL_MANUAL:
LOAD
    Num#(КХЕ_текст) as КХЕ,
    Pick(Match(Num#(КХЕ_текст), 1160, 1250, 1270, 1120, 1320, 1300),

// 1240: В ФАЙЛЕ «Ручники.xlsx» ПОДПИСИ ПЕРЕПУТАНЫ (проверено по файлу)!
// Ряд '1240 постов'   (янв-25 = 128) = КОЛ-ВО ОБЪЕКТОВ по эталону → знаменатель RPU
// Ряд '1240 объектов' (165–174)      = посты, в RPU НЕ используются
// Независимое подтверждение: парсер приложений (ГРУПП4+Эгида) за апр-2026 = 124,
// и в ручнике ряд 'постов' за апр-2026 тоже = 124.
// ВАЖНО: NoConcatenate ОБЯЗАТЕЛЕН — набор полей совпадает с KOL_MANUAL,
// иначе Qlik автоконкатенирует KOL_1240 В KOL_MANUAL,
// и RESIDENT KOL_1240 ниже падает с "Table 'KOL_1240' not found".
KOL_1240:
NoConcatenate
LOAD
    1240 as КХЕ,
    If(WildMatch(КХЕ_текст, '*пост*'),   'Кол-во объектов',
    If(WildMatch(КХЕ_текст, '*объект*'), 'Кол-во постов')) as Показатель,
    Год, Месяц,
    Sum(Количество) as Количество
RESIDENT RAW_MANUAL
WHERE WildMatch(КХЕ_текст, '*1240*')
GROUP BY КХЕ_текст, Год, Месяц;

// ============================================================
// АМОРТИЗАЦИЯ (2025 Excel + 2026 QVD)
// В файле 2025 и, возможно, в QVD 2026 оборудование охраны сидит
// под КХЕ 1245 (в файле 1240 НЕТ) → ре-мап в FACT_KPI (см. ниже).
// КХЕ загружается «как есть», без слияния 1270.1 → 1270.
// ============================================================
AMORT_RAW:
LOAD
	Num#(Replace(Trim(KHE), ',', '.'))                   as KHE_n,
    Num#(Год)                                            as Год,
    If(Num#(Месяц) > 12, Num(Month(Месяц)), Num#(Месяц)) as Месяц,
    Num#(Амортизация, '#.##0,00', ',', ' ')              as AMORT_VAL
FROM [$(vAmortLib)/амортизация 2025.xlsx]
(ooxml, embedded labels)
WHERE Len(Trim(KHE)) > 0;

CONCATENATE (AMORT_RAW)
LOAD
	Num#(Replace(Trim(KHE), ',', '.'))                   as KHE_n,
    Num#(Год)                                            as Год,
    Num#(Месяц)                                          as Месяц,
    AMORTZAPERIOD                                        as AMORT_VAL
FROM [lib://QVD Tables/KaspiService/Amort_2026.qvd] (qvd)
WHERE Len(Trim(KHE)) > 0;

// ============================================================
// FACT_OPEX — проводки
// 1245 (ПЦН/тех.охрана) включаем и ре-мапим на 1240:
// эталон 1240 = 1240 ∪ 1245. Исходный КХЕ сохраняем в [КХЕ исх]
// (нужен для разводки классов по поставщику в FACT_OPEX_TMP).
// ============================================================
FACT_OPEX:
LOAD
    If(Num#(Trim([КХЕ факт])) = 1245, 1240, Num#(Trim([КХЕ факт]))) as [КХЕ факт],
    Num#(Trim([КХЕ факт]))                      as [КХЕ исх],
    If(Num#(Trim([КХЕ факт])) = 1245, 1240, Num#(Trim([КХЕ факт])))
      & '|' & MonthStart([Дата документа])      as %KEY_KHE_PERIOD,
    Trim([Наименование])                        as [Наименование КХЕ факт],
    Num#(Trim([Код статьи факт]))               as [Код статьи],
    Trim([Наименование затрат])                 as [Наименование затрат],
    Date([Дата документа], 'DD.MM.YYYY')        as [Дата документа OPEX],
    Year([Дата документа])                      as [Год OPEX],
    Month([Дата документа])                     as [Месяц OPEX],
    Date(MonthStart([Дата документа]), 'MMM-YYYY') as [МесяцГод OPEX],
    MonthStart([Дата документа])                as [ДатаСортировки OPEX],
    Trim([Поставщик])                           as Поставщик,
    Num#(PurgeChar(PurgeChar(Trim([Сумма (Факт)] & ''), ' '), Chr(160)), '#0,00', ',', '') as [Сумма факт]
FROM [lib://amortizacia_2025 (hq_tolybayev_59417)/проводки.xlsx]
(ooxml, embedded labels)
WHERE Len(Trim([КХЕ факт])) > 0
  AND IsNum([Дата документа])
  AND Year([Дата документа]) >= 2025
  AND Match(Num#(Trim([КХЕ факт])), 501,502,503,902,1120,1140,1160,1240,1245,1250,1270,1300,1320);

// Класс по умолчанию: «Прочие»; для 1270 — «Прочие расходы»; для 1300 — исключается из итога.
// Правила 1240∪1245 (по [КХЕ исх], ДО ремапа; проверены по проводкам — эталон сходится до тиын):
//   1245 + 2131 + ТОО "Жедел Кузет"      → Охрана (ПЦН)   (янв-25 = 41 450 529)
//   1245 + 2131(прочие)/2209/2213/2303   → Тех охрана     (янв-25 = 10 654 829)
//   1245 + 2711/2712                     → Командировки   (1240+1245 янв-25 = 962 081)
// Коды 1245 «1» и 2603 → «Прочие» (вопрос заказчику: оставить в Прочих или в Тех охрану).
FACT_OPEX_TMP:
NoConcatenate
LOAD *,
    If([КХЕ исх] = 1245 and [Код статьи] = 2131
         and WildMatch(Поставщик, '*Жедел*Кузет*'), 'Охрана (ПЦН)',
    If([КХЕ исх] = 1245 and Match([Код статьи], 2131, 2209, 2213, 2303), 'Тех охрана',
    If([КХЕ исх] = 1245 and Match([Код статьи], 2711, 2712), 'Командировки',
       If(Len(Trim(Класс)) = 0,
          Pick(Match([КХЕ факт], 1300, 1270) + 1,
               'Прочие', 'Исключить', 'Прочие расходы'),
          Класс))))                                      as [Класс расходов]
RESIDENT FACT_OPEX;
DROP TABLE FACT_OPEX;
RENAME TABLE FACT_OPEX_TMP TO FACT_OPEX;

// ============================================================
// RPU — витрина расхода на единицу
// ВНИМАНИЕ: метрика «Расход на охрану 1 АТМ» ВЫВОДИТСЯ КАК КХЕ 1245
// (ремарка заказчика). Здесь и далее %КХЕ = витринный код (1245),
// а все расходные/амортизационные мапы читаются по ИСХОДНОМУ КХЕ (1240).
// ============================================================
RPU:
LOAD
    If(ApplyMap('MAP_METRIC', КХЕ & '|' & Показатель, 'НЕТ') = 'Расход на охрану 1 АТМ', 1245, КХЕ) as [КХЕ RPU],
    If(ApplyMap('MAP_METRIC', КХЕ & '|' & Показатель, 'НЕТ') = 'Расход на охрану 1 АТМ', 1245, КХЕ) as %КХЕ,
    Показатель                                                   as [Показатель RPU],
    ApplyMap('MAP_METRIC', КХЕ & '|' & Показатель, 'НЕТ')        as Метрика,
    Год                                                          as [Год RPU],
    Месяц                                                        as [Месяц RPU],
    МесяцГод                                                     as [МесяцГод RPU],
    ДатаСортировки                                               as [ДатаСортировки RPU],
    Количество                                                   as Знаменатель,
    RangeSum(ApplyMap('MAP_AMORT',      КХЕ & '|' & Num(ДатаСортировки), 0)) as Амортизация,
    // у 1245 (АТМ) «Всего OPEX» = ПЦН: в блоке 1245 показываем только охрану ПЦН
    If(ApplyMap('MAP_METRIC', КХЕ & '|' & Показатель, 'НЕТ') = 'Расход на охрану 1 АТМ',
       RangeSum(ApplyMap('MAP_OPEX_PCN',   КХЕ & '|' & Num(ДатаСортировки), 0)),
       RangeSum(ApplyMap('MAP_OPEX_TOTAL', КХЕ & '|' & Num(ДатаСортировки), 0))) as [Всего OPEX],
    RangeSum(
        ApplyMap('MAP_AMORT',      КХЕ & '|' & Num(ДатаСортировки), 0),
        ApplyMap('MAP_OPEX_TOTAL', КХЕ & '|' & Num(ДатаСортировки), 0)
    )                                                            as Расход,
    RangeSum(ApplyMap('MAP_OPEX_SVYAZ',     КХЕ & '|' & Num(ДатаСортировки), 0)) as Связь,
    RangeSum(ApplyMap('MAP_OPEX_RASH',      КХЕ & '|' & Num(ДатаСортировки), 0)) as Расходники,
    RangeSum(ApplyMap('MAP_OPEX_OBSL',      КХЕ & '|' & Num(ДатаСортировки), 0)) as Обслуживание,
    RangeSum(ApplyMap('MAP_OPEX_TRIP',      КХЕ & '|' & Num(ДатаСортировки), 0)) as Командировки,
    RangeSum(ApplyMap('MAP_OPEX_PROCHIE',   КХЕ & '|' & Num(ДатаСортировки), 0)) as Прочие,
    RangeSum(ApplyMap('MAP_OPEX_OBSL_KP',   КХЕ & '|' & Num(ДатаСортировки), 0)) as [Обслуживание КП],
    RangeSum(ApplyMap('MAP_OPEX_RASH_KP',   КХЕ & '|' & Num(ДатаСортировки), 0)) as [Расходники КП],
    RangeSum(ApplyMap('MAP_OPEX_PROCH_KP',  КХЕ & '|' & Num(ДатаСортировки), 0)) as [Прочие расходы],
    RangeSum(ApplyMap('MAP_OPEX_POCHTA',    КХЕ & '|' & Num(ДатаСортировки), 0)) as Почтовые,
    RangeSum(ApplyMap('MAP_OPEX_INFO',      КХЕ & '|' & Num(ДатаСортировки), 0)) as Информационные,
    RangeSum(ApplyMap('MAP_OPEX_FIZ_OHR',   КХЕ & '|' & Num(ДатаСортировки), 0)) as [Физ охрана],
    RangeSum(ApplyMap('MAP_OPEX_TEH_OHR',   КХЕ & '|' & Num(ДатаСортировки), 0)) as [Тех охрана],
    RangeSum(ApplyMap('MAP_OPEX_STOJANKA',  КХЕ & '|' & Num(ДатаСортировки), 0)) as Стоянки,
    RangeSum(ApplyMap('MAP_OPEX_PCN',       КХЕ & '|' & Num(ДатаСортировки), 0)) as [Охрана (ПЦН)],
    RangeSum(ApplyMap('MAP_OPEX_TRANSPORT', КХЕ & '|' & Num(ДатаСортировки), 0)) as [Расходы на транспортировку],
    RangeSum(ApplyMap('MAP_OPEX_DOSTAVKA',  КХЕ & '|' & Num(ДатаСортировки), 0)) as [Доставка АТМ],
    RangeSum(ApplyMap('MAP_OPEX_REMONT',    КХЕ & '|' & Num(ДатаСортировки), 0)) as Ремонт,
    RangeSum(ApplyMap('MAP_OPEX_GSM',       КХЕ & '|' & Num(ДатаСортировки), 0)) as ГСМ
RESIDENT FACT_KPI
WHERE Показатель <> 'Амортизация'
  AND ApplyMap('MAP_METRIC', КХЕ & '|' & Показатель, 'НЕТ') <> 'НЕТ';

// расход на единицу; спец-формулы:
//   1140 «на 1 конверт» — знаменатель/2;
//   1240 «охрана объекта» — расход БЕЗ ПЦН; 1245 «охрана 1 АТМ» — только ПЦН
RPU_T1:
NoConcatenate
LOAD *,
    If(Знаменатель > 0,
       If(Метрика = 'Расход на 1 конверт',
          Расход / (Знаменатель / 2),
       If(Метрика = 'Расход на охрану 1 объекта',
          (Амортизация + [Всего OPEX] - [Охрана (ПЦН)]) / Знаменатель,
       If(Метрика = 'Расход на охрану 1 АТМ',
          [Охрана (ПЦН)] / Знаменатель,
          Расход / Знаменатель))),
       null())                                  as [Расход на единицу]
RESIDENT RPU;
DROP TABLE RPU;
RENAME TABLE RPU_T1 TO RPU;

// % изменения месяц-к-месяцу
RPU_T2:
NoConcatenate
LOAD *,
    If(%КХЕ = Peek('%КХЕ')
       and Метрика = Peek('Метрика')
       and Peek('Расход на единицу') > 0,
       ([Расход на единицу] - Peek('Расход на единицу')) / Peek('Расход на единицу'),
       null()
    ) as [% изменения RPU]
RESIDENT RPU
ORDER BY %КХЕ, Метрика, [ДатаСортировки RPU];
DROP TABLE RPU;
RENAME TABLE RPU_T2 TO RPU;

// ============================================================
// RPU_6M — таблица «Расход на единицу за 6 месяцев» (по примеру заказчика)
// Читает RPU и LAYOUT, ничего в них не меняет.
// Связь с моделью — ТОЛЬКО по %КХЕ: остальные поля изолированы
// (уникальные имена → синтетических ключей нет).
//
// П2 = 6 последних ЗАКРЫТЫХ месяцев (Max по RPU; период обрезан в FACT_KPI
//      условием < MonthStart(Today()) — неполный месяц не затекает).
// П1 = те же 6 месяцев со сдвигом v6M_Offset (12 = год назад, 6 = пред. полугодие).
// «Расход на единицу за период» = Σ (Аморт+OPEX) / среднее помесячное кол-во;
// спец-формулы 1-в-1 как в RPU_T1: 1140 конверт (знаменатель/2),
//   1240 объекты (без ПЦН), 1245 АТМ (только ПЦН).
// Статья «на единицу» = Σ статьи за период / среднее кол-во.
// Выдаются ВСЕ метрики (вкл. вторичные); в сводном листе лишние прячутся set-фильтром в UI.
// % = П2/П1 − 1, цвет ▲▼ — в чарте.
// ============================================================

// --- границы периодов ---
R6M_MAXD:
LOAD Max([ДатаСортировки RPU]) as MAXD RESIDENT RPU;
LET v6M_Max = Num(Peek('MAXD'));
DROP TABLE R6M_MAXD;
LET v6M_Offset  = 12;
LET v6M_P2_From = Num(YearStart($(v6M_Max)));
LET v6M_P1_From = Num(YearStart(AddMonths($(v6M_Max), -12)));
LET v6M_P1_To   = Num(MonthEnd(AddMonths($(v6M_Max), -12)));

// --- подписи единиц измерения (дословно из листа «Единица» примера) ---

// --- помесячная база: КХЕ × Метрика × месяц + все классы OPEX ---
R6M_BASE:
NoConcatenate
LOAD
    %КХЕ, Метрика,
    Num([ДатаСортировки RPU])  as ДС,
    Знаменатель,
    Амортизация, [Всего OPEX],
    Связь, Расходники, Обслуживание, Командировки, Прочие,
    [Обслуживание КП], [Расходники КП], [Прочие расходы],
    Почтовые, Информационные,
    [Физ охрана], [Тех охрана], Стоянки, [Охрана (ПЦН)],
    [Расходы на транспортировку], [Доставка АТМ], Ремонт, ГСМ
RESIDENT RPU;

// разворачиваем статьи макета (только этой КХЕ) — по %КХЕ
JOIN (R6M_BASE)
LOAD
    КХЕ_L  as %КХЕ,
    Статья as Статья6M,
    ПорядокСтатьи
RESIDENT LAYOUT;

// --- лонг: значение каждой статьи за месяц + метка периода ---
R6M_LONG:
NoConcatenate
LOAD
    %КХЕ, Метрика, Статья6M, ПорядокСтатьи, Знаменатель,
    If(ДС >= YearStart($(v6M_Max)),
       'П2',
       If(ДС >= YearStart(AddMonths($(v6M_Max), -12))
          and ДС <= MonthEnd(AddMonths($(v6M_Max), -12)),
          'П1')) as Период6M,
    Pick(
        Match(Статья6M,
            'Амортизация','Связь','Расходники','Обслуживание','Командировки','Прочие','Всего OPEX',
            'Обслуживание КП','Расходники КП','Прочие расходы',
            'Почтовые','Информационные',
            'Физ охрана','Тех охрана','Стоянки','Охрана (ПЦН)',
            'Расходы на транспортировку','Доставка АТМ',
            'Ремонт','ГСМ'),
        Амортизация, Связь, Расходники, Обслуживание, Командировки, Прочие, [Всего OPEX],
        [Обслуживание КП], [Расходники КП], [Прочие расходы],
        Почтовые, Информационные,
        [Физ охрана], [Тех охрана], Стоянки, [Охрана (ПЦН)],
        [Расходы на транспортировку], [Доставка АТМ],
        Ремонт, ГСМ
    ) as СуммаСтатьи
RESIDENT R6M_BASE;
DROP TABLE R6M_BASE;

// --- знаменатели по периодам: среднее помесячное кол-во ---
// строго одна строка на месяц (статья 'Всего OPEX' есть у каждой КХЕ),
// иначе Avg по раздутым JOIN с LAYOUT строкам искажает среднее
R6M_KOL:
LOAD
    %КХЕ, Метрика,
    Avg(If(Период6M = 'П1', Знаменатель)) as K1,
    Avg(If(Период6M = 'П2', Знаменатель)) as K2
RESIDENT R6M_LONG
WHERE Len(Trim(Период6M)) > 0
  AND Статья6M = 'Всего OPEX'
GROUP BY %КХЕ, Метрика;

// --- агрегат по периодам: суммы статей ---
R6M_AGG:
LOAD
    %КХЕ, Метрика, Статья6M, ПорядокСтатьи,
    Sum(If(Период6M = 'П1', СуммаСтатьи))  as S1,
    Sum(If(Период6M = 'П2', СуммаСтатьи))  as S2
RESIDENT R6M_LONG
WHERE Len(Trim(Период6M)) > 0
GROUP BY %КХЕ, Метрика, Статья6M, ПорядокСтатьи;
DROP TABLE R6M_LONG;

// --- знаменатели к статьям ---
LEFT JOIN (R6M_AGG)
LOAD %КХЕ, Метрика, K1, K2 RESIDENT R6M_KOL;
DROP TABLE R6M_KOL;

// --- метрический агрегат для строки «Расход на единицу» (вкл. ПЦН для 1245) ---
R6M_MET:
LOAD
    %КХЕ, Метрика,
    Sum(If(Статья6M = 'Амортизация',  S1)) as A1,
    Sum(If(Статья6M = 'Амортизация',  S2)) as A2,
    Sum(If(Статья6M = 'Всего OPEX',   S1)) as O1,
    Sum(If(Статья6M = 'Всего OPEX',   S2)) as O2,
    Sum(If(Статья6M = 'Охрана (ПЦН)', S1)) as P1,
    Sum(If(Статья6M = 'Охрана (ПЦН)', S2)) as P2,
    Max(K1)                                as K1,
    Max(K2)                                as K2
RESIDENT R6M_AGG
GROUP BY %КХЕ, Метрика;

// --- статьи «на единицу» (без «Всего OPEX» — как в примере) ---
RPU_6M:
NoConcatenate
LOAD
    %КХЕ,
    Метрика                                                  as [Метрика 6M],
    ApplyMap('MAP_UNIT_NAME', %КХЕ & '|' & Метрика, Метрика) as [Единица измерения],
    ApplyMap('MAP_KHE_ORDER', %КХЕ, 99)                      as [Порядок КХЕ],
    ApplyMap('MAP_KHE_ORDER', %КХЕ, 99) * 10
      + ApplyMap('MAP_METRIC_ORDER', %КХЕ & '|' & Метрика, 1) as [Сорт метрики],
    Статья6M                                                 as Строка,
    'Статья'                                                 as [Тип строки],
    10 + ПорядокСтатьи                                       as [Порядок строки],
    If(K1 > 0, S1 / K1)                                      as Значение_П1,
    If(K2 > 0, S2 / K2)                                      as Значение_П2,
    ''                                                       as [Причина 6M]
RESIDENT R6M_AGG
WHERE Статья6M <> 'Всего OPEX';

// --- строка «Среднее кол-во» (вверху блока) ---
CONCATENATE (RPU_6M)
LOAD DISTINCT
    %КХЕ,
    Метрика                                                  as [Метрика 6M],
    ApplyMap('MAP_UNIT_NAME', %КХЕ & '|' & Метрика, Метрика) as [Единица измерения],
    ApplyMap('MAP_KHE_ORDER', %КХЕ, 99)                      as [Порядок КХЕ],
    ApplyMap('MAP_KHE_ORDER', %КХЕ, 99) * 10
      + ApplyMap('MAP_METRIC_ORDER', %КХЕ & '|' & Метрика, 1) as [Сорт метрики],
    'Среднее кол-во — '
      & ApplyMap('MAP_KOL_LABEL', %КХЕ & '|' & Метрика, 'кол-во') as Строка,
    'Количество'                                             as [Тип строки],
    2                                                        as [Порядок строки],
    K1                                                       as Значение_П1,
    K2                                                       as Значение_П2,
    ''                                                       as [Причина 6M]
RESIDENT R6M_AGG;

// --- строка «Расход на единицу» ---
CONCATENATE (RPU_6M)
LOAD
    %КХЕ,
    Метрика                                                  as [Метрика 6M],
    ApplyMap('MAP_UNIT_NAME', %КХЕ & '|' & Метрика, Метрика) as [Единица измерения],
    ApplyMap('MAP_KHE_ORDER', %КХЕ, 99)                      as [Порядок КХЕ],
    ApplyMap('MAP_KHE_ORDER', %КХЕ, 99) * 10
      + ApplyMap('MAP_METRIC_ORDER', %КХЕ & '|' & Метрика, 1) as [Сорт метрики],
    'Расход на единицу'                                      as Строка,
    'Итог'                                                   as [Тип строки],
    1                                                        as [Порядок строки],
    If(K1 > 0,
       If(Метрика = 'Расход на 1 конверт',
          RangeSum(A1, O1) / (K1 / 2),
       If(Метрика = 'Расход на охрану 1 объекта',
          (RangeSum(A1, O1) - RangeSum(P1, 0)) / K1,
       If(Метрика = 'Расход на охрану 1 АТМ',
          RangeSum(P1, 0) / K1,
          RangeSum(A1, O1) / K1))))                          as Значение_П1,
    If(K2 > 0,
       If(Метрика = 'Расход на 1 конверт',
          RangeSum(A2, O2) / (K2 / 2),
       If(Метрика = 'Расход на охрану 1 объекта',
          (RangeSum(A2, O2) - RangeSum(P2, 0)) / K2,
       If(Метрика = 'Расход на охрану 1 АТМ',
          RangeSum(P2, 0) / K2,
          RangeSum(A2, O2) / K2))))                          as Значение_П2,
    ''                                                       as [Причина 6M]
RESIDENT R6M_MET;

            + ApplyMap('MAP_METRIC_ORDER', %КХЕ & '|' & Метрика, 1)),
        ApplyMap('MAP_KHE_ORDER', %КХЕ, 99) * 1000
      + ApplyMap('MAP_METRIC_ORDER', %КХЕ & '|' & Метрика, 1) * 100
      + 91
    )                                                          as Статья,
    [Год RPU], [Месяц RPU],
    Dual([МесяцГод RPU], Num([ДатаСортировки RPU]))            as [МесяцГод RPU],
    [ДатаСортировки RPU],
    [Расход на единицу]                                        as Значение
RESIDENT RPU;

// строка «%»
CONCATENATE (RPU_LONG)
LOAD
    %КХЕ,
    Dual([КХЕ RPU], ApplyMap('MAP_KHE_ORDER', %КХЕ, 99))       as [КХЕ RPU],
    Dual(Метрика,
         ApplyMap('MAP_KHE_ORDER', %КХЕ, 99) * 10
       + ApplyMap('MAP_METRIC_ORDER', %КХЕ & '|' & Метрика, 1)) as Метрика,
    Dual(
        '%'
          & Repeat(Chr(160),
              ApplyMap('MAP_KHE_ORDER', %КХЕ, 99) * 3
            + ApplyMap('MAP_METRIC_ORDER', %КХЕ & '|' & Метрика, 1)),
        ApplyMap('MAP_KHE_ORDER', %КХЕ, 99) * 1000
      + ApplyMap('MAP_METRIC_ORDER', %КХЕ & '|' & Метрика, 1) * 100
      + 92
    )                                                          as Статья,
    [Год RPU], [Месяц RPU],
    Dual([МесяцГод RPU], Num([ДатаСортировки RPU]))            as [МесяцГод RPU],
    [ДатаСортировки RPU],
    [% изменения RPU]                                          as Значение
RESIDENT RPU;

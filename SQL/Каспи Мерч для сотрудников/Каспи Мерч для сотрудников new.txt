select
    q.sku,
    q.product,
    CONCAT(q.product, ' ', ROW_NUMBER() OVER (PARTITION BY q.product ORDER BY sku)) AS new_product
      from (select
         		t.sku,
         		t.product,
                t.merchant,
                count() over(partition by t.sku) cnt_sku,
                count() over(partition by t.product) cnt_product
             from BI.stock_merch_v t
             where upper(t.code) like upper('%Kaspi%')
                   and t.merchant = 'Ucon store') q
    where 1 = 1
    and q.cnt_sku = 1
    and q.cnt_product > 1;

TEMP_PRICE_CHANGE:
LOAD 
    NEW_PRODUCT_NAME, 
    PRODUCT_PRICE as UniqProductPrice,
    CATEGORY,
    DATE(MIN(ORDERDATE)) as MinDate, 
    DATE(MAX(ORDERDATE)) as MaxDate,
    AutoNumber(CATEGORY & '|' & PRODUCT_PRICE) as PriceKey,
    MAX(ORDERDATE) - MIN(ORDERDATE) AS period
Resident ORDERS_FINAL
GROUP BY 
    NEW_PRODUCT_NAME, 
    PRODUCT_PRICE,
    CATEGORY;

MAP_START_DATES:
MAPPING LOAD 
    NEW_PRODUCT_NAME & '|' & CATEGORY, 
    MIN(MinDate) 
Resident TEMP_PRICE_CHANGE 
GROUP BY NEW_PRODUCT_NAME, CATEGORY;

MAP_MIN_PRICES:
MAPPING LOAD 
    NEW_PRODUCT_NAME & '|' & CATEGORY, 
    MIN(UniqProductPrice) 
Resident TEMP_PRICE_CHANGE 
GROUP BY NEW_PRODUCT_NAME, CATEGORY;

MAP_SECOND_MIN_PRICES:
MAPPING LOAD 
    NEW_PRODUCT_NAME & '|' & CATEGORY, 
    FirstSortedValue(UniqProductPrice, UniqProductPrice, 2) // Берем 2-е значение в сортировке цен
Resident TEMP_PRICE_CHANGE
GROUP BY NEW_PRODUCT_NAME, CATEGORY;


PRICE_CHANGE:
LOAD
    *,
    IF(MinDate = ApplyMap('MAP_START_DATES', NEW_PRODUCT_NAME & '|' & CATEGORY) AND period <= 1, 
        'Стартовая цена',
        IF(Year(MinDate) > 2024 and period < 21  AND 
           (UniqProductPrice = ApplyMap('MAP_MIN_PRICES', NEW_PRODUCT_NAME & '|' & CATEGORY)),
            'Акционная цена',
            
            // ПРИОРИТЕТ 3: Минимальная цена (если держалась долго, >= 21 дня)
            IF(UniqProductPrice = ApplyMap('MAP_MIN_PRICES', NEW_PRODUCT_NAME & '|' & CATEGORY), 
                'Минимальная цена',
                
                'Стандартная цена'
            )
        )
    ) as PriceCategory

LOAD 
     DATE($(V_MINDATE_N) + rowno() - 1) AS TempDate
AUTOGENERATE 1
While $(V_MINDATE_N)+IterNo() <= $(V_MAXDATE);

ORDERDATE_CALENDAR:
Load
	Date(TempDate,'DD.MM.YYYY') as ORDERDATE,
    Year(TempDate) as ORDER_YEAR,
    Month(TempDate) as ORDER_MONTH,
    Day(TempDate) as ORDER_DAY,
    MonthName(TempDate) as ORDER_MONTHYEAR,
    WeekDay(TempDate) as ORDER_WEEKDAY,
    Date(WeekStart(TempDate),'YYYY.MM.DD')&'-'&Date(WeekEnd(TempDate),'YYYY.MM.DD') as ORDER_WEEK
RESIDENT TempCalendar
ORDER BY TempDate ASC;

drop table TempCalendar;

CLICKDATA:
LOAD
MonthName(dt) as DATE_CLICK,
uniq_users,
new_users,
add_cart_users,
add_cartcta_users,
buy_now_click_users,
transaction_users
;
SQL
WITH t1 as (SELECT 
	DATE_TRUNC('month', approximatearrivaltimestamp) AS dt, 
	count(distinct user_id_hash) AS uniq_users, 
	COUNT(CASE WHEN new_users = 1 THEN 1 END) as new_users
FROM
(SELECT approximatearrivaltimestamp, user_id_hash, ROW_NUMBER() OVER(PARTITION BY user_id_hash ORDER BY approximatearrivaltimestamp) as new_users
from S0173.AWS_EVENTS_N 
where approximatearrivaltimestamp >= '2024-09-01'
and event_name='shop_product_teaser_click'
and brand= 'Kaspi') a
GROUP BY 1),
t2 as (SELECT DATE_TRUNC('month', approximatearrivaltimestamp) AS dt, count(distinct user_id_hash) AS add_cart_users
from S0173.AWS_EVENTS_N 
where approximatearrivaltimestamp >= '2024-09-01'
and event_name='shop_add_to_cart_click'
and brand= 'Kaspi'
GROUP BY 1),
t3 as (SELECT DATE_TRUNC('month', approximatearrivaltimestamp) AS dt, count(distinct user_id_hash) AS add_cartcta_users
from S0173.AWS_EVENTS_N 
where approximatearrivaltimestamp >= '2024-09-01'
and event_name='shop_product_cta_click'
and brand= 'Kaspi'
GROUP BY 1),
t6 as (SELECT DATE_TRUNC('month', approximatearrivaltimestamp) AS dt, count(distinct user_id_hash) AS buy_now_click_users
from S0173.AWS_EVENTS_N 
where approximatearrivaltimestamp >= '2024-09-01'
and event_name='shop_buy_now_click'
and brand= 'Kaspi'
GROUP BY 1),
t7 as (SELECT DATE_TRUNC('month', approximatearrivaltimestamp) AS dt, count(distinct user_id_hash) AS transaction_users
from S0173.AWS_EVENTS_N 
where approximatearrivaltimestamp >= '2024-09-01'
and event_name='shop_product_transaction'
and brand= 'Kaspi'
GROUP BY 1)
SELECT t1.dt, uniq_users, new_users, add_cart_users, add_cartcta_users, buy_now_click_users ,transaction_users
FROM t1 
LEFT JOIN t2 ON t1.dt = t2.dt 
LEFT JOIN t3 ON t1.dt = t3.dt 
LEFT JOIN t6 ON t1.dt = t6.dt
LEFT JOIN t7 ON t1.dt = t7.dt;

Temp_Orders:
LOAD 
    MonthName(ORDERDATE) AS DATE_CLICK, 
    Count(DISTINCT If(PAYED_PRICE > 0 AND Match(current_status_status, 'completed', 'acceptedByHub', 'uploadedToPostomat', 'receivedByHub'), ORDERCODE, NULL())) AS TotalOrders,
    Sum(If(PAYED_PRICE > 0 AND Match(current_status_status, 'completed', 'acceptedByHub', 'uploadedToPostomat', 'receivedByHub'), QUANTITY, 0)) AS TOTAL_QUANTITY
RESIDENT ORDERS_FINAL 
GROUP BY MonthName(ORDERDATE);

;

[PRODUCT_CLICK]:
Load
   MonthName(dt) as DATE_CLICK,
   product_id as product_id_view,
   product_name as product_name_view,
   VIEW_KEY,
   click_cnt as PRODUCT_CLICK_ACTIONS
;
SQL
WITH t1 AS(select 
  category_id, 
  created_time, 
  product_id,
  product_name,
  date_trunc('month', approximatearrivaltimestamp) AS dt,
  count(distinct user_id_hash) click_cnt
from S0173.AWS_EVENTS_N 
where approximatearrivaltimestamp >= '2024-09-01'
and event_name='shop_product_teaser_click'
and brand= 'Kaspi'
and created_time is not null
and price is not null
group by 1,2,3,4,5)
SELECT 
	dt, 
  	product_id,
  	product_name, 
  	CONCAT(a.category_level_2,'|',DATE_FORMAT(t1.created_time, '%d.%m.%Y')) as VIEW_KEY,
  	click_cnt
FROM t1 left join  
(SELECT category_id, category_level_2 
from S0173.AWS_EVENTS_N 
where approximatearrivaltimestamp >= '2024-09-01'
and event_name='shop_product_teaser_click'
and brand= 'Kaspi'
and created_time is not null
and price is not null
and`up#selected_language` in ('ru-RU')
group by 1,2) a ON t1.category_id = a.category_id
WHERE a.category_level_2 NOT REGEXP '[ӘҒҚҢӨҰҮҺІәғқңөұүһі]'
;

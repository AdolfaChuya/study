WITH Step1_Clean AS (
    SELECT r.C_NUM
         , u.c_name
         , r.C_ARHIV_RET_REASON                  
         , r.c_put_budget 
         , UPPER(REPLACE(u.c_email, ' ', '')) AS C_EMAIL
         -- Очищаем ВСЮ строку от переносов (не отрезаем Причина:)
         , REPLACE(REPLACE(r.C_ARHIV_RET_REASON, CHR(13), ''), CHR(10), '') AS full_sms
           
    FROM s06.z#kas_request_pay r
       , s06.z#product p          
       , s06.z#user u             
    WHERE p.id = r.id 
      AND u.id = p.c_create_user
      AND r.STATE_ID != 'PAID'
      AND r.C_DATE_REQ > TO_DATE('01/01/2024', 'DD/MM/YYYY')
),
Step2_Positions AS (
    SELECT C_NUM
         , c_name
         , C_ARHIV_RET_REASON
         , c_put_budget
         , C_EMAIL
         , full_sms
         -- Считаем сколько всего дат в тексте
         , REGEXP_COUNT(full_sms, '[0-9]{2}\.[0-9]{2}\.[0-9]{4} [0-9]{2}:[0-9]{2}:[0-9]{2}') AS date_count
         -- Позиция ПОСЛЕДНЕЙ даты
         , CASE 
             WHEN REGEXP_COUNT(full_sms, '[0-9]{2}\.[0-9]{2}\.[0-9]{4} [0-9]{2}:[0-9]{2}:[0-9]{2}') > 0 
             THEN REGEXP_INSTR(full_sms, '[0-9]{2}\.[0-9]{2}\.[0-9]{4} [0-9]{2}:[0-9]{2}:[0-9]{2}', 1, 
                  REGEXP_COUNT(full_sms, '[0-9]{2}\.[0-9]{2}\.[0-9]{4} [0-9]{2}:[0-9]{2}:[0-9]{2}'))
             ELSE 0 
           END AS last_date_pos
    FROM Step1_Clean
),
Step3_Extract AS (
    SELECT C_NUM
         , c_name
         , C_ARHIV_RET_REASON
         , c_put_budget
         , C_EMAIL
         , full_sms
         , date_count
         , last_date_pos
         -- Текст ДО последней даты
         , CASE WHEN last_date_pos > 0 THEN SUBSTR(full_sms, 1, last_date_pos - 1) ELSE NULL END AS text_before_date
         -- Текст ПОСЛЕ последней даты (19 символов самой даты)
         , CASE WHEN last_date_pos > 0 THEN SUBSTR(full_sms, last_date_pos + 19) ELSE NULL END AS text_after_date
         -- Позиция ПОСЛЕДНЕГО "Причина:" перед последней датой (ищем с конца)
         , CASE WHEN last_date_pos > 1 THEN INSTR(SUBSTR(full_sms, 1, last_date_pos - 1), 'Причина:', -1) ELSE 0 END AS prev_prichina_pos
    FROM Step2_Positions
)
SELECT C_NUM
     , c_name
     , C_ARHIV_RET_REASON
     , c_put_budget
     , C_EMAIL
     
     -- 1. ДАТА/ВРЕМЯ (последняя дата)
     , CASE 
         WHEN last_date_pos > 0 THEN SUBSTR(full_sms, last_date_pos, 19)
         ELSE NULL 
       END AS RET_DATE_TIME
     
     -- 2. СТАТУС "ИЗ"
     , TRIM(
         CASE 
           WHEN last_date_pos > 0 THEN
             CASE 
               -- "Из статуса:" есть в последнем блоке (после последнего "Причина:" перед датой)
               WHEN prev_prichina_pos > 0 
                    AND INSTR(SUBSTR(text_before_date, prev_prichina_pos), 'Из статуса:"') > 0 THEN 
                   REGEXP_SUBSTR(SUBSTR(text_before_date, prev_prichina_pos), 'Из статуса:"([^"]+)"', 1, 1, 'i', 1)
               -- "Из статуса:" есть, но "Причина:" не было (первый блок = последний)
               WHEN prev_prichina_pos = 0 
                    AND INSTR(text_before_date, 'Из статуса:"') > 0 THEN
                   REGEXP_SUBSTR(text_before_date, 'Из статуса:"([^"]+)"', 1, 1, 'i', 1)
               -- Нет "Из статуса:" — пытаемся извлечь имя статуса (текст после "Причина:" и до даты)
               WHEN prev_prichina_pos > 0 THEN
                   TRIM(REGEXP_SUBSTR(SUBSTR(text_before_date, prev_prichina_pos + LENGTH('Причина:')), '([А-ЯЁ][а-яё ]+)$'))
               -- Нет ни "Из статуса:", ни "Причина:" — берём весь текст до даты
               ELSE TRIM(text_before_date)
             END
           ELSE NULL 
         END
       ) AS RET_STATUS_FROM

     -- 3. КТО ВЕРНУЛ (текст после даты, до "В статус:" или "Причина:")
     , TRIM(
         CASE 
           WHEN last_date_pos > 0 THEN 
             CASE 
               -- Режем до "В статус:" если есть
               WHEN INSTR(text_after_date, 'В статус:') > 0 THEN 
                   SUBSTR(text_after_date, 1, INSTR(text_after_date, 'В статус:') - 1)
               -- Иначе режем до "Причина:" если есть
               WHEN INSTR(text_after_date, 'Причина:') > 0 THEN 
                   SUBSTR(text_after_date, 1, INSTR(text_after_date, 'Причина:') - 1)
               -- Иначе берём весь хвост
               ELSE text_after_date
             END
           ELSE NULL 
         END
       ) AS RET_PERSON
       
     -- 4. СТАТУС "В"
     , TRIM(
         CASE 
           WHEN last_date_pos > 0 AND INSTR(text_after_date, 'В статус:"') > 0 THEN 
               REGEXP_SUBSTR(text_after_date, 'В статус:"([^"]+)"', 1, 1, 'i', 1)
           ELSE NULL 
         END
       ) AS RET_STATUS_TO

FROM Step3_Extract;

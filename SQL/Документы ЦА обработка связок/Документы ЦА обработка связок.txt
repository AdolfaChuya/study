Date_Bounds:
LOAD
    Min([День]) as MinDate,
    Max([День]) as MaxDate
RESIDENT ANL;

LET vMinDate = Num(Peek('MinDate', 0, 'Date_Bounds'));
LET vMaxDate = Num(Peek('MaxDate', 0, 'Date_Bounds'));
DROP TABLE Date_Bounds;


Calendar_Base:
LOAD
    $(vMinDate) + RowNo() - 1 as DateNum
AUTOGENERATE ($(vMaxDate) - $(vMinDate) + 1);


MasterCalendar:
LOAD
    Date(DateNum, 'YYYY-MM-DD')                   as [День], 
    Day(DateNum)                                  as [День месяца],
    WeekDay(DateNum)                              as [День недели],
    
    Week(DateNum)                                 as [Номер недели],
    WeekYear(DateNum)                             as [Год недели],
    WeekYear(DateNum) & '-W' & Num(Week(DateNum), '00') as [Неделя], 
    Year(DateNum)                                 as [Год],
    InYearToDate(DateNum, Today(), 0)             as [YTD_Флаг]
RESIDENT Calendar_Base
WHERE Num(WeekDay(DateNum)) < 5; // Пн, Вт, Ср, Чт, Пт


DROP TABLE Calendar_Base;
